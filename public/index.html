How to automate email sending with Google Apps Script

  1. Open the Google Sheet you configured in your .env file (GOOGLE_SHEET_ID).
  2. Go to Extensions → Apps Script in the Sheet's menu bar.
  3. Delete any existing code and paste the function below.
  4. Click Save (💾), then click Run to authorize the script (grant Gmail and Sheets permissions when prompted).
  5. Go to Triggers (⏰ clock icon in the left sidebar) → click + Add Trigger.
  6. Set: Function = processScheduledEmails, Event source = Time-driven, Type = Minutes timer, Interval = Every 1 minute.
  7. Click Save. The script will now check for due emails every minute and send them automatically — even when your PC is off!
Apps Script — processScheduledEmails 
function processScheduledEmails() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var now = new Date();

  for (var i = 1; i < data.length; i++) {  // Skip header row
    var eventId  = data[i][0];  // Column A: event_id
    var toEmail  = data[i][1];  // Column B: to_email
    var subject  = data[i][2];  // Column C: subject
    var body     = data[i][3];  // Column D: body (HTML)
    var sendAt   = new Date(data[i][4]);  // Column E: send_at
    var status   = data[i][5];  // Column F: status

    if (status === 'scheduled' && sendAt <= now) {
      try {
        GmailApp.sendEmail(toEmail, subject, '', { htmlBody: body });
        sheet.getRange(i + 1, 6).setValue('sent');
        sheet.getRange(i + 1, 7).setValue(now.toISOString());
        Logger.log('Sent email to ' + toEmail + ' for event: ' + eventId);
      } catch (e) {
        sheet.getRange(i + 1, 6).setValue('error: ' + e.message);
        Logger.log('Failed for event ' + eventId + ': ' + e.message);
      }
    }
  }
}
